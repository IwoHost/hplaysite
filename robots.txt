# The words and the screenshots are public and there is nothing to hide from a crawler.
# The photographs of a person are a different matter: they are on the page for anyone who
# reads it, and that is not the same as wanting them in an image search for years. Blocking
# the folder is the documented way to keep pictures out of Google Images.
User-agent: *
Allow: /
Disallow: /img/photos/
